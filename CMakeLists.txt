cmake_minimum_required(VERSION 3.20)
project(ewrtc VERSION 0.2.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

include(cmake/EWRTCOptions.cmake)
include(cmake/EWRTCComponents.cmake)
include(cmake/EWRTCDependencies.cmake)

foreach(c IN LISTS all_components)
  if(NOT c IN_LIST enabled)
    continue()
  endif()
  add_library(ewrtc_${c}_objects OBJECT ${src_${c}})
  target_include_directories(ewrtc_${c}_objects PRIVATE include src)
  target_compile_definitions(ewrtc_${c}_objects PRIVATE ${definitions})
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ewrtc_${c}_objects PRIVATE -Wall -Wextra -Wpedantic)
    if(EWRTC_WARNINGS_AS_ERRORS)
      target_compile_options(ewrtc_${c}_objects PRIVATE -Werror)
    endif()
  endif()
  target_link_libraries(ewrtc_${c}_objects PRIVATE ${${c}_libs})
  add_library(ewrtc_${c} STATIC $<TARGET_OBJECTS:ewrtc_${c}_objects>)
  if(c STREQUAL "pal_linux")
    # Keep the platform provider independently linkable without public helper targets.
    target_sources(ewrtc_${c} PRIVATE $<TARGET_OBJECTS:ewrtc_pal_objects> $<TARGET_OBJECTS:ewrtc_common_objects>)
  endif()
  target_include_directories(ewrtc_${c} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  foreach(d IN LISTS deps_${c})
    if(NOT c STREQUAL "pal_linux")
      target_link_libraries(ewrtc_${c} PUBLIC ewrtc_${d})
    endif()
  endforeach()
  target_link_libraries(ewrtc_${c} PUBLIC ${${c}_libs})
  if(NOT c STREQUAL "pal_linux")
    list(APPEND aggregate_objects $<TARGET_OBJECTS:ewrtc_${c}_objects>)
    list(APPEND aggregate_libs ${${c}_libs})
  endif()
endforeach()
if(session IN_LIST enabled)
  add_library(ewrtc STATIC ${aggregate_objects})
  add_library(ewrtc::ewrtc ALIAS ewrtc)
  target_include_directories(ewrtc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(ewrtc PUBLIC ${aggregate_libs})
  install(TARGETS ewrtc EXPORT ewrtcTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()

include(cmake/EWRTCInstall.cmake)
include(cmake/EWRTCTests.cmake)
include(cmake/EWRTCExamples.cmake)
