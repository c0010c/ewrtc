cmake_minimum_required(VERSION 3.20)
project(ewrtc_external_consumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
find_package(ewrtc CONFIG REQUIRED COMPONENTS ${COMPONENT})
foreach(internal common pal crypto sdp stun turn ice dtls srtp rtp media)
  if(TARGET ewrtc::${internal})
    message(FATAL_ERROR "Internal target leaked: ewrtc::${internal}")
  endif()
endforeach()
# Optional protocol component requests must also be reported as unavailable.
find_package(ewrtc CONFIG REQUIRED OPTIONAL_COMPONENTS stun srtp)
if(ewrtc_stun_FOUND OR ewrtc_srtp_FOUND)
  message(FATAL_ERROR "Internal protocols remain available package components")
endif()
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE ewrtc::${COMPONENT})
if(NOT COMPONENT STREQUAL "pal_linux")
  target_compile_definitions(consumer PRIVATE HAVE_SESSION=1)
endif()
if(TARGET ewrtc::pal_linux)
  target_link_libraries(consumer PRIVATE ewrtc::pal_linux)
  target_compile_definitions(consumer PRIVATE HAVE_LINUX=1)
endif()
# Every installed header must be usable on its own, including from C++.
get_target_property(public_include ewrtc::${COMPONENT} INTERFACE_INCLUDE_DIRECTORIES)
list(GET public_include 0 public_include)
file(GLOB_RECURSE public_headers RELATIVE "${public_include}" "${public_include}/*.h")
foreach(header IN LISTS public_headers)
  string(MAKE_C_IDENTIFIER "${header}" name)
  foreach(language c cpp)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.${language}" "#include <${header}>\n")
    add_library(header_${name}_${language} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${name}.${language}")
    target_link_libraries(header_${name}_${language} PRIVATE ewrtc::${COMPONENT})
  endforeach()
endforeach()
